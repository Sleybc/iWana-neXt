const rows = [];
const canary = new Map();
for (const l of readFileSync(`${S}/corrida.ndjson`, 'utf8').trim().split('\n')) {
  const r = JSON.parse(l);
  if (r.step !== 'canary') continue;
  const k = `${r.data.table}.${r.data.column}`;
  if (r.data.row.startsWith('sonda')) continue;
  canary.set(k, canary.get(k) === 'sobrevive' || r.data.survived ? 'sobrevive' : 'desaparece');
}
for (const l of readFileSync(`${S}/cols-all.txt`, 'utf8').trim().split('\n')) {
  const [t, c, type, max, nullable] = l.split('|');
  const k = `${t}.${c}`;
  const hit = group.find(([re]) => re.test(t));
  let cls, why;
  if (hit?.[1] === 'LISTADA-MOD14') [cls, why] = [hit[1], hit[2]];
  else if (rules.has(k)) [cls, why] = ['CUBIERTA', rules.get(k)];
  else if (res[k]) [cls, why] = ['RESIDUO', res[k]];
  else if (ex[k] === 'CODIGO-CUBIERTA') [cls, why] = ['CUBIERTA', 'UserAccountErasureParticipant (correo sintético)'];
  else if (ex[k]) [cls, why] = ['EXCLUIDA', ex[k]];
  else if (hit) [cls, why] = [hit[1], hit[2]];
  else if (type === 'USER-DEFINED') [cls, why] = ['EXCLUIDA', 'ENUM: dominio cerrado'];
  else if (/(^|_)(status|type|code|number|key|hash|hmac|token|secret|sku|segment|channel|method|result|cycle|priority|role|version|currency|country|treatment|kind|origin|source_context|origin_context|queue_name)$/.test(c)
    || /(_id|_ids|_ref|_key|_context)$/.test(c)) [cls, why] = ['EXCLUIDA', 'CODIGO: referencia, código o dominio del sistema'];
  else [cls, why] = ['REVISAR', ''];
  rows.push({ t, c, type, max, nullable, cls, why, can: canary.get(k) ?? 'no sembrada' });
}
writeFileSync(OUT, ['tabla,columna,tipo,largo,nulable,clase,motivo_o_dueno,canario'].concat(
  rows.map((r) => [r.t, r.c, r.type, r.max, r.nullable, r.cls, `"${r.why}"`, r.can].join(','))).join('\n') + '\n');
const count = {};
for (const r of rows) count[r.cls] = (count[r.cls] ?? 0) + 1;
console.log(count);
for (const r of rows.filter((x) => x.cls === 'REVISAR')) console.log(`${r.t}.${r.c}:${r.type}:${r.max}`);
// Coherencia: toda CUBIERTA sembrada desaparece; todo RESIDUO sembrado sobrevive.
console.log('incoherencias', rows.filter((r) => (r.cls === 'CUBIERTA' && r.can === 'sobrevive') || (r.cls === 'RESIDUO' && r.can === 'desaparece')).map((r) => `${r.t}.${r.c}`));
console.log('residuos no sembrados', rows.filter((r) => r.cls === 'RESIDUO' && r.can === 'no sembrada').map((r) => `${r.t}.${r.c}`));
