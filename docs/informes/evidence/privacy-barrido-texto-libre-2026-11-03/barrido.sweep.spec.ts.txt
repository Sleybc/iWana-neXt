  async function columnsOf(table: string): Promise<Col[]> {
    return qr.query(
      `SELECT column_name, data_type, udt_name, character_maximum_length AS max,

    const r = await qr.query(
      `SELECT e.enumlabel FROM pg_enum e JOIN pg_type t ON t.oid = e.enumtypid

  /** Inserta una fila con un canario en cada columna textual libre. `row` = etiqueta de la sonda. */
  async function seed(
    table: string,
    row: string,
    over: Record<string, unknown> = {},
  ): Promise<string> {
    const cols = await columnsOf(table);
    if (cols.length === 0) throw new Error(`Tabla inexistente ${table}`);
    const values = new Map<string, unknown>();
    const casts = new Map<string, string>();
    for (const c of cols) {
      if (c.generated) continue;
      const name = c.column_name;
      if (name in over) {
        values.set(name, over[name]);
        continue;
      }
      if (name === 'tenant_id') {
        values.set(name, tenantId);
        continue;
      }
      const required = !c.nullable && !c.has_default;
      const text = ['character varying', 'text', 'character'].includes(c.data_type);
      if (text) {
        if (/_encrypted$/.test(name)) {
          values.set(name, encryptAes256Gcm(tag(table, name, row, 'cifrada'), key));
          continue;
        }
        const free = !REF.test(name) && !STRUCT.test(name) && (c.max === null || c.max >= 7);
        if (free) {
          values.set(name, tag(table, name, row, 'texto'));
          continue;
        }
        if (required) {
          values.set(name, c.max !== null && c.max < 7 ? 'X' : `S${randomUUID().slice(0, 5)}`);
        }
        continue;
      }
      if (c.data_type === 'jsonb' || c.data_type === 'json') {
        if (required || !STRUCT.test(name)) {
          values.set(name, JSON.stringify({ nota: tag(table, name, row, 'json') }));
          casts.set(name, c.data_type);
        }
        continue;
      }
      if (!required) continue;
      if (c.data_type === 'USER-DEFINED') {
        values.set(name, await firstEnum(c.udt_name));
        casts.set(name, `${q}."${c.udt_name}"`);
      } else if (c.data_type === 'uuid') values.set(name, randomUUID());
      else if (c.data_type.startsWith('timestamp') || c.data_type === 'date') {
        values.set(name, new Date().toISOString());
      } else if (
        ['numeric', 'integer', 'smallint', 'bigint', 'double precision'].includes(c.data_type)
      ) {
        values.set(name, 1);
      } else if (c.data_type === 'boolean') values.set(name, false);
      else if (c.data_type === 'inet') values.set(name, `10.9.${seq % 250}.${(seq * 7) % 250}/32`);
      else if (c.data_type === 'ARRAY') values.set(name, []);
      else throw new Error(`Tipo sin relleno ${table}.${name}: ${c.data_type}`);
    }
    for (const [name, value] of Object.entries(over)) {
      const c = cols.find((x) => x.column_name === name);
      if (!c) throw new Error(`Columna inexistente ${table}.${name}`);
      if (c.data_type === 'USER-DEFINED') casts.set(name, `${q}."${c.udt_name}"`);
      if ((c.data_type === 'jsonb' || c.data_type === 'json') && typeof value !== 'string') {
        values.set(name, JSON.stringify(value));
        casts.set(name, c.data_type);
      }
    }
    for (let attempt = 0; attempt < 25; attempt += 1) {
      const names = [...values.keys()];
      const placeholders = names.map(
        (n, i) => `$${i + 1}${casts.has(n) ? `::${casts.get(n)}` : ''}`,
      );
      const sql = `INSERT INTO ${q}."${table}" (${names.map((n) => `"${n}"`).join(', ')})
        VALUES (${placeholders.join(', ')}) RETURNING *`;
      try {
        await qr.query('SAVEPOINT s');
        const r = await qr.query(
          sql,
          names.map((n) => values.get(n)),
        );
        await qr.query('RELEASE SAVEPOINT s');
        return String(r[0].id ?? r[0].contract_id ?? '');
      } catch (e) {
        await qr.query('ROLLBACK TO SAVEPOINT s');
        const err = e as { code?: string; constraint?: string; message: string };
        if (err.code === '23514' && err.constraint) {
          const found = await qr.query(
            `SELECT pg_get_constraintdef(oid) AS d FROM pg_constraint
              WHERE conname = $1 AND connamespace = $2::regnamespace LIMIT 1`,
            [err.constraint, SCHEMA],
          );
          const def = String(found[0]?.d);
          const touched = names.filter(
            (n) => !(n in over) && n !== 'tenant_id' && new RegExp(`\\b${n}\\b`).test(def),
          );
          if (touched.length === 0) throw new Error(`CHECK ${err.constraint} en ${table}: ${def}`);
          const literal = /'([^']+)'/.exec(def)?.[1];
          for (const n of touched) {
            const c = cols.find((x) => x.column_name === n) as Col;
            const idx = canaries.findIndex(
              (k) => k.table === table && k.column === n && k.row === row,
            );
            if (idx >= 0) canaries.splice(idx, 1);
            skipped.push({ table, column: n, row, constraint: err.constraint });
            if (!c.nullable && !c.has_default) values.set(n, literal ?? 'X');
            else values.delete(n);
          }
          continue;
        }
        throw new Error(`Siembra ${table}: ${err.code} ${err.message}`);
      }
    }
    throw new Error(`Siembra ${table}: demasiados reintentos`);
  }

          `SELECT "${c.column}"::text AS v FROM ${q}."${c.table}" WHERE "${c.column}" IS NOT NULL`,
        );
        survived = all.some((r: { v: string }) => {
          try {
            return decryptAes256Gcm(r.v, key) === c.code;
          } catch {
            return false;
          }
        });
      } else {
        const rows = await boot.query(
          `SELECT 1 FROM ${q}."${c.table}" WHERE "${c.column}"::text LIKE $1`,
          [`%${c.code}%`],
        );
        survived = rows.length > 0;
      }
      log('canary', { ...c, survived });
    }
    log(
      'audit-sanitize',
      sanitizeAuditPayload({
        reason: 'R',
        notes: 'N',
        note: 'N',
        body: 'B',
        subject: 'S',
        comment: 'C',
        description: 'D',
        title: 'T',
        summary: 'U',
      }),
    );
  }, 280000);
});
